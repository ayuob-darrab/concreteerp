## 1️⃣ سكربت النسخ الاحتياطي

**احفظ كـ:** `backup.ps1`

```powershell
# ═══════════════════════════════════════════════════════════
# سكربت النسخ الاحتياطي لـ ConcreteERP
# ═══════════════════════════════════════════════════════════

$ErrorActionPreference = "Stop"
$date = Get-Date -Format "yyyy-MM-dd_HH-mm"
$projectPath = "C:\laragon\www\ConcreteERP"
$backupPath = "C:\laragon\backups"

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan
Write-Host "           بدء النسخ الاحتياطي لـ ConcreteERP              " -ForegroundColor Cyan
Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

# إنشاء مجلد النسخ الاحتياطية
if (!(Test-Path $backupPath)) {
    New-Item -ItemType Directory -Path $backupPath -Force | Out-Null
    Write-Host "✅ تم إنشاء مجلد النسخ الاحتياطية" -ForegroundColor Green
}

# نسخ المشروع
Write-Host "`n📦 جاري نسخ المشروع..." -ForegroundColor Yellow
$projectBackup = "$backupPath\ConcreteERP_backup_$date"
Copy-Item -Path $projectPath -Destination $projectBackup -Recurse -Force
Write-Host "✅ تم نسخ المشروع إلى: $projectBackup" -ForegroundColor Green

# نسخ ملفات التكوين
Write-Host "`n📄 جاري نسخ ملفات التكوين..." -ForegroundColor Yellow
Set-Location $projectPath
Copy-Item ".env" ".env.backup_$date" -Force
Copy-Item "composer.json" "composer.json.backup_$date" -Force
if (Test-Path "composer.lock") {
    Copy-Item "composer.lock" "composer.lock.backup_$date" -Force
}
Write-Host "✅ تم نسخ ملفات التكوين" -ForegroundColor Green

# إنشاء Git tag
Write-Host "`n🏷️ جاري إنشاء Git tag..." -ForegroundColor Yellow
git add . 2>$null
git commit -m "Backup before upgrade - $date" 2>$null
git tag -a "backup-$date" -m "Backup before Laravel upgrade" 2>$null
Write-Host "✅ تم إنشاء Git tag: backup-$date" -ForegroundColor Green

Write-Host "`n═══════════════════════════════════════════════════════════" -ForegroundColor Cyan
Write-Host "           ✅ اكتمل النسخ الاحتياطي بنجاح!                 " -ForegroundColor Green
Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

Write-Host "`n📍 مواقع النسخ الاحتياطية:" -ForegroundColor White
Write-Host "   - المشروع: $projectBackup" -ForegroundColor Gray
Write-Host "   - Git Tag: backup-$date (محلي فقط)" -ForegroundColor Gray
Write-Host "`n⚠️ ملاحظة: كل شيء محفوظ محلياً - لم يتم الرفع على GitHub" -ForegroundColor Yellow
```

## 2️⃣ سكربت الترقية إلى Laravel 9

**احفظ كـ:** `upgrade-to-9.ps1`

```powershell
# ═══════════════════════════════════════════════════════════
# سكربت الترقية إلى Laravel 9
# ═══════════════════════════════════════════════════════════

$ErrorActionPreference = "Stop"
$projectPath = "C:\laragon\www\ConcreteERP"

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan
Write-Host "           بدء الترقية إلى Laravel 9                       " -ForegroundColor Cyan
Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

Set-Location $projectPath

# التحقق من النسخة الاحتياطية
$confirm = Read-Host "`n⚠️ هل قمت بأخذ نسخة احتياطية؟ (y/n)"
if ($confirm -ne "y") {
    Write-Host "❌ يرجى أخذ نسخة احتياطية أولاً!" -ForegroundColor Red
    exit
}

# تحديث composer.json
Write-Host "`n📝 جاري تحديث composer.json..." -ForegroundColor Yellow

$composerJson = Get-Content "composer.json" -Raw | ConvertFrom-Json

# تحديث الإصدارات
$composerJson.require.php = "^8.0"
$composerJson.require.'laravel/framework' = "^9.0"
$composerJson.require.'laravel/sanctum' = "^3.0"
$composerJson.require.'laravel/tinker' = "^2.7"
$composerJson.require.'laravel/ui' = "^4.0"

# إزالة الحزم القديمة
$composerJson.require.PSObject.Properties.Remove('fruitcake/laravel-cors')

# تحديث require-dev
$composerJson.'require-dev'.'nunomaduro/collision' = "^6.1"
$composerJson.'require-dev'.PSObject.Properties.Remove('facade/ignition')

$composerJson | ConvertTo-Json -Depth 10 | Set-Content "composer.json"
Write-Host "✅ تم تحديث composer.json" -ForegroundColor Green

# حذف vendor و lock
Write-Host "`n🗑️ جاري حذف vendor و composer.lock..." -ForegroundColor Yellow
Remove-Item -Recurse -Force vendor -ErrorAction SilentlyContinue
Remove-Item composer.lock -ErrorAction SilentlyContinue
Write-Host "✅ تم الحذف" -ForegroundColor Green

# تثبيت الحزم
Write-Host "`n📦 جاري تثبيت الحزم..." -ForegroundColor Yellow
composer install

# إضافة spatie/laravel-ignition
Write-Host "`n📦 جاري إضافة spatie/laravel-ignition..." -ForegroundColor Yellow
composer require spatie/laravel-ignition --dev

# مسح الـ cache
Write-Host "`n🧹 جاري مسح الـ cache..." -ForegroundColor Yellow
php artisan optimize:clear

# التحقق من الإصدار
Write-Host "`n📋 التحقق من الإصدار..." -ForegroundColor Yellow
php artisan --version

Write-Host "`n═══════════════════════════════════════════════════════════" -ForegroundColor Cyan
Write-Host "           ✅ اكتملت الترقية إلى Laravel 9!                " -ForegroundColor Green
Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan
Write-Host "`n⚠️ يرجى اختبار المشروع قبل المتابعة للمرحلة التالية" -ForegroundColor Yellow
```

## 5️⃣ سكربت التراجع

**احفظ كـ:** `rollback.ps1`

```powershell
# ═══════════════════════════════════════════════════════════
# سكربت التراجع عن الترقية
# ═══════════════════════════════════════════════════════════

$ErrorActionPreference = "Stop"
$projectPath = "C:\laragon\www\ConcreteERP"

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Red
Write-Host "           ⚠️ التراجع عن الترقية                           " -ForegroundColor Red
Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Red

Set-Location $projectPath

# عرض الـ tags المتاحة
Write-Host "`n📋 الـ Tags المتاحة:" -ForegroundColor Yellow
git tag -l

$tag = Read-Host "`nأدخل اسم الـ tag للتراجع إليه"

if ([string]::IsNullOrEmpty($tag)) {
    Write-Host "❌ لم يتم إدخال tag!" -ForegroundColor Red
    exit
}

# التأكيد
$confirm = Read-Host "`n⚠️ هل أنت متأكد من التراجع إلى $tag؟ (y/n)"
if ($confirm -ne "y") {
    Write-Host "❌ تم الإلغاء" -ForegroundColor Red
    exit
}

# التراجع
Write-Host "`n🔄 جاري التراجع..." -ForegroundColor Yellow
git stash
git checkout $tag

# إعادة تثبيت الحزم
Write-Host "`n📦 جاري إعادة تثبيت الحزم..." -ForegroundColor Yellow
Remove-Item -Recurse -Force vendor -ErrorAction SilentlyContinue
composer install

# مسح الـ cache
Write-Host "`n🧹 جاري مسح الـ cache..." -ForegroundColor Yellow
php artisan optimize:clear

# التحقق
Write-Host "`n📋 التحقق من الإصدار..." -ForegroundColor Yellow
php artisan --version

Write-Host "`n═══════════════════════════════════════════════════════════" -ForegroundColor Green
Write-Host "           ✅ تم التراجع بنجاح!                             " -ForegroundColor Green
Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Green
```
